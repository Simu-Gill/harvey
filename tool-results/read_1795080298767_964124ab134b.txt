   135→function collectionHealth(rate: number) {
   136→  if (rate >= 0.9) return { label: "Strong", tone: "good" as const, icon: CheckCircle2 };
   137→  if (rate >= 0.75) return { label: "Steady", tone: "info" as const, icon: CheckCircle2 };
   138→  if (rate >= 0.55) return { label: "Watch", tone: "watch" as const, icon: Clock };
   139→  return { label: "At Risk", tone: "risk" as const, icon: TriangleAlert };
   140→}
   141→
   142→function portfolioHealth(statusMix: DashboardDTO["statusMix"]) {
   143→  const balance = (s: string) => statusMix.find((x) => x.status === s)?.balance ?? 0;
   144→  const count = (s: string) => statusMix.find((x) => x.status === s)?.count ?? 0;
   145→  const total = statusMix.reduce((s, x) => s + x.balance, 0);
   146→  if (total <= 0) return { label: "No Book", tone: "info" as const, icon: CircleDashed };
   147→  const overdueShare = balance("overdue") / total;
   148→  const behindShare = balance("behind") / total;
   149→  if (overdueShare > 0.15 || count("overdue") >= 3)
   150→    return { label: "At Risk", tone: "risk" as const, icon: TriangleAlert };
   151→  if (overdueShare > 0.05 || behindShare > 0.3)
   152→    return { label: "Watch", tone: "watch" as const, icon: Clock };
   153→  return { label: "Healthy", tone: "good" as const, icon: CheckCircle2 };
   154→}

   156→/** Round, human-friendly axis ticks (0, 4k, 8k, 12k …). */
   157→export function niceTicks(max: number, count = 4): number[] {
   158→  if (max <= 0) return [0];
   159→  const raw = max / count;
   160→  const mag = Math.pow(10, Math.floor(Math.log10(raw)));
   161→  const norm = raw / mag;
   162→  const step = (norm <= 1 ? 1 : norm <= 2 ? 2 : norm <= 5 ? 5 : 10) * mag;
   163→  const ticks: number[] = [];
   164→  for (let v = 0; v <= max + step * 0.01; v += step) ticks.push(Math.round(v * 100) / 100);
   165→  return ticks;
   166→}
